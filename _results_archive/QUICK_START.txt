⚡ COMANDO RÁPIDO — Python:

import tensorflow as tf
import json
import h5py

# Carregar modelo
model = tf.keras.models.load_model('Redes Neurais/results/models/cnn1d_tag_auth_best.keras')

# Carregar threshold
with open('Redes Neurais/results/models/cnn1d_threshold.json') as f:
    config = json.load(f)
    threshold = config['cnn_threshold']

# Carregar dados
with h5py.File('Redes Neurais/results/data/dataset_cnn_yeq_0_30dB.h5', 'r') as f:
    X_test = f['test/y_eq'][:]
    y_test = f['test/y'][:]

# Fazer predição
pred = model.predict(X_test)
y_pred = (pred >= threshold).astype(int)

print(f"Acurácia: {(y_pred.flatten() == y_test).mean():.4f}")
